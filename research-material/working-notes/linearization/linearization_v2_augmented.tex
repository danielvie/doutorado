\title{Linearization of an Affine Switched System}

Consider the affine switched system
\begin{equation}
    \dot{x}(t)=A_{\sigma(t)}x(t)+b_{\sigma(t)},
    \qquad x(t)\in\R^n.
    \tag{1}
\end{equation}
The switching signal is piecewise constant.  Its mode order is fixed,
while its switching times may vary:
\begin{equation}
    \sigma(t)=q_i=\bar\sigma(\bar t_{i-1}),
    \qquad t_{i-1}\leq t<t_i,
    \qquad i=1,\ldots,N.
    \tag{2}
\end{equation}

Introduce the augmented state and matrix
\begin{equation}
    \X(t)=\begin{bmatrix}x(t)\\1\end{bmatrix},
    \qquad
    \widetilde A_q=
    \begin{bmatrix}A_q&b_q\\0&0\end{bmatrix}.
\end{equation}
Then the affine system becomes the homogeneous system
\begin{equation}
    \boxed{\dot{\X}(t)=\widetilde A_{\sigma(t)}\X(t)}.
\end{equation}



For each interval, define
\begin{equation}
    \Delta t_i=t_i-t_{i-1},
    \qquad
    \Delta\bar t_i=\bar t_i-\bar t_{i-1},
    \qquad
    \Delta t_i=\Delta\bar t_i+\delta t_i,
    \qquad
    F_i=\widetilde A_{q_i}.
\end{equation}
The state transition on interval $i$ is
\begin{equation}
    \X(t_i)=e^{F_i\Delta t_i}\X(t_{i-1}).
\end{equation}
Therefore,
\begin{equation}
    \X(t_N)=
    e^{F_N\Delta t_N} e^{F_{N-1}\Delta t_{N-1}}\cdots e^{F_1\Delta t_1}\X(t_0).
    \label{eq:actual}
\end{equation}
The first interval is the rightmost factor.

Define the nominal transition matrix for interval $i$ by
\begin{equation}
    \phi_i=e^{F_i\Delta\bar t_i}.
    \label{eq:transition-definitions}
\end{equation}



Let
\begin{equation}
    e(t)=\X(t)-\bar\X(t),
    \qquad
    \X(t_0)=\bar\X(t_0)+e(t_0).
\end{equation}
Using \eqref{eq:actual}, the exact terminal error is
\begin{equation}
\begin{aligned}
    e(t_N)={}&
    e^{F_N(\Delta\bar t_N+\delta t_N)}
    e^{F_{N-1}(\Delta\bar t_{N-1}+\delta t_{N-1})}
    \cdots \\
    &\quad e^{F_1(\Delta\bar t_1+\delta t_1)}
    \bigl(\bar\X(t_0)+e(t_0)\bigr)
    -\phi_N\phi_{N-1}\cdots\phi_1\bar\X(t_0).
\end{aligned}
\label{eq:exact-error}
\end{equation}

For a constant matrix $F_i$, expand $e^{F_i s}$ around
$s=\Delta\bar t_i$:
\begin{equation}
\begin{aligned}
    \frac{\mathrm{d}}{\mathrm{d}s}e^{F_i s}&=F_i e^{F_i s},\\
    e^{F_i(\Delta\bar t_i+\delta t_i)}
    &=e^{F_i\Delta\bar t_i}
      +F_i e^{F_i\Delta\bar t_i}\,\delta t_i
      +\mathcal O(\delta t_i^2)\\
    &=\phi_i+F_i\phi_i\,\delta t_i
      +\mathcal O(\delta t_i^2)\\
    &\doteq\phi_i+F_i\phi_i\,\delta t_i.
\end{aligned}
\label{eq:single-transition-linearization}
\end{equation}
In the following, $\doteq$ denotes equality up to first order in the
duration perturbations: terms containing two or more factors
$\delta t_i$ are discarded.
Substituting the first-order approximation into every transition gives
\begin{equation}
\begin{aligned}
&e^{F_N(\Delta\bar t_N+\delta t_N)}
 e^{F_{N-1}(\Delta\bar t_{N-1}+\delta t_{N-1})}
 \cdots
 e^{F_1(\Delta\bar t_1+\delta t_1)} \\
&\quad\doteq
 (\phi_N+F_N\phi_N\,\delta t_N)
 (\phi_{N-1}+F_{N-1}\phi_{N-1}\,\delta t_{N-1})
 \cdots
 (\phi_1+F_1\phi_1\,\delta t_1) \\
&\quad\doteq
 \phi_N\phi_{N-1}\cdots\phi_1 \\
&\qquad{}+F_N\phi_N\phi_{N-1}\cdots\phi_1\,\delta t_N \\
&\qquad{}+\phi_NF_{N-1}\phi_{N-1}\cdots\phi_1\,\delta t_{N-1} \\
&\qquad{}+\cdots \\
&\qquad{}+\phi_N\phi_{N-1}\cdots\phi_2F_1\phi_1\,\delta t_1.
\end{aligned}
\label{eq:product-linearization}
\end{equation}
Substituting \eqref{eq:product-linearization} into
\eqref{eq:exact-error}, cancelling the nominal term, and discarding
$\delta t_i e(t_0)$ yields
\begin{equation}
\begin{aligned}
    e(t_N)\doteq{}&
    \phi_N\phi_{N-1}\cdots\phi_1e(t_0) \\
    &+\phi_N\phi_{N-1}\cdots\phi_2F_1\phi_1\bar\X(t_0)\,\delta t_1 \\
    &+\phi_N\phi_{N-1}\cdots\phi_3F_2\phi_2\phi_1\bar\X(t_0)\,\delta t_2 \\
    &+\cdots \\
    &+F_N\phi_N\phi_{N-1}\cdots\phi_1\bar\X(t_0)\,\delta t_N.
\end{aligned}
\label{eq:sum-linearization}
\end{equation}

Collect the duration perturbations and their sensitivity columns:
\begin{equation}
\begin{aligned}
    \delta\mathbf t&=
    \begin{bmatrix}
        \delta t_1\\ \delta t_2\\ \vdots\\ \delta t_N
    \end{bmatrix}, \\
    \Gamma=\Bigl[\,&
        \phi_N\phi_{N-1}\cdots\phi_2F_1\phi_1\bar\X(t_0), \\
        &\phi_N\phi_{N-1}\cdots\phi_3F_2\phi_2\phi_1\bar\X(t_0),
        \ \ldots, \\
        &F_N\phi_N\phi_{N-1}\cdots\phi_1\bar\X(t_0)
    \Bigr].
\end{aligned}
\end{equation}
For the final compact form, define the complete nominal transition
\begin{equation}
    \Phi=\phi_N\phi_{N-1}\cdots\phi_1.
\end{equation}
Equation \eqref{eq:sum-linearization} is thus
\begin{equation}
    \boxed{
        e(t_N)\doteq\Phi e(t_0)+\Gamma\,\delta\mathbf t
    }.
    \label{eq:final}
\end{equation}
